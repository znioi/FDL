# Experiment 8

## Aim

Using LSTM for prediction of future weather of cities in Python.

## Description

Long Short-Term Memory (LSTM) networks are a specialized type of Recurrent Neural Network (RNN) designed to handle sequential and time-dependent data. They are capable of learning long-term dependencies using memory cells and gating mechanisms, making them well-suited for time series forecasting tasks such as weather prediction.

In this experiment, we apply an LSTM-based approach to analyze and prepare weather data for temperature prediction. The dataset is loaded from a CSV file containing historical weather information, including date and temperature values. Proper preprocessing steps such as date formatting, sorting, and normalization are performed to make the data suitable for sequence modeling.

The temperature data is scaled using MinMaxScaler and converted into sequences using a sliding window approach, where a fixed number of past observations, such as 60 days, are used to predict future values. These sequences are reshaped into a format compatible with LSTM input.

Although this implementation focuses on data preprocessing and visualization, it forms the foundation for building an LSTM model that can learn temporal patterns in weather data and make accurate future temperature predictions.

## Algorithm

1. Import required Python libraries for data handling, visualization, and LSTM modeling.
2. Read the weather dataset from CSV file.
3. Convert date column to datetime and set it as index.
4. Select the temperature column.
5. Normalize data using MinMaxScaler.
6. Define window size, such as 60.
7. Create input-output sequences.
8. Reshape data for LSTM.
9. Define LSTM model with stacked LSTM, Dropout, and Dense layers.
10. Compile using Adam optimizer and train the model.
11. Predict values on entire dataset and inverse scale results.
12. Plot actual vs predicted temperature.
13. Predict next day temperature using last window data.

## Code

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout

file_path = "/content/Project1WeatherDataset.csv"

try:
    df_weather = pd.read_csv(file_path)
    print(f"Dataset loaded successfully from {file_path}. Shape: {df_weather.shape}")
    print("First 5 rows of the dataset:")
    print(df_weather.head())
    print("\nDataset columns:")
    print(df_weather.columns)
except FileNotFoundError:
    print(f"Error: The file '{file_path}' was not found. Please upload it or check the path.")
    df_weather = None

if df_weather is not None:
    date_column = "Date"
    temperature_column = "Temp_C"

    if date_column in df_weather.columns:
        df_weather[date_column] = pd.to_datetime(df_weather[date_column])
        df_weather = df_weather.set_index(date_column)
        df_weather = df_weather.sort_index()
    else:
        print(f"Warning: Date column '{date_column}' not found. Skipping date indexing.")

    if temperature_column in df_weather.columns:
        data = df_weather[[temperature_column]].values
    else:
        print(f"Error: Temperature column '{temperature_column}' not found. Please check your column name.")
        data = None

    if data is not None:
        scaler = MinMaxScaler(feature_range=(0, 1))
        scaled_data = scaler.fit_transform(data)

        window_size = 60
        train_len = len(scaled_data) * 0.8
        train_data = scaled_data[0 : int(train_len), :]

        x_train, y_train = [], []

        for i in range(window_size, len(train_data)):
            x_train.append(train_data[i - window_size : i, 0])
            y_train.append(train_data[i, 0])

        x_train, y_train = np.array(x_train), np.array(y_train)
        x_train = np.reshape(x_train, (x_train.shape[0], x_train.shape[1], 1))

        print(f"\nShape of x_train: {x_train.shape}")
        print(f"Shape of y_train: {y_train.shape}")

        plt.figure(figsize=(12, 6))
        plt.plot(df_weather[temperature_column])
        plt.title(f"Kaggle Weather Data: {temperature_column} History")
        plt.ylabel(temperature_column)
        plt.xlabel("Date")
        plt.show()
    else:
        print("Data preparation failed due to missing temperature column.")
else:
    print("Data loading failed. Please check the file path and try again.")

if "x_train" in locals() and "y_train" in locals() and x_train.size > 0:
    model = Sequential(
        [
            LSTM(50, return_sequences=True, input_shape=(x_train.shape[1], 1)),
            Dropout(0.2),
            LSTM(50, return_sequences=False),
            Dropout(0.2),
            Dense(25),
            Dense(1),
        ]
    )

    model.compile(optimizer="adam", loss="mean_squared_error")
    history = model.fit(
        x_train,
        y_train,
        batch_size=32,
        epochs=20,
        validation_split=0.1,
        verbose=1,
    )

    plt.figure(figsize=(10, 6))
    plt.plot(history.history["loss"], label="Training Loss")
    plt.plot(history.history["val_loss"], label="Validation Loss")
    plt.title("Model Loss Over Epochs")
    plt.xlabel("Epoch")
    plt.ylabel("Loss")
    plt.legend()
    plt.show()
else:
    print("Skipping model building and training: x_train or y_train not available or empty.")

if "model" in locals() and "scaled_data" in locals() and scaled_data is not None and scaled_data.size > 0:
    x_all = []

    for i in range(window_size, len(scaled_data)):
        x_all.append(scaled_data[i - window_size : i, 0])

    x_all = np.array(x_all)
    x_all = np.reshape(x_all, (x_all.shape[0], x_all.shape[1], 1))

    all_predictions = model.predict(x_all)
    all_predictions = scaler.inverse_transform(all_predictions)

    predicted_plot = np.empty_like(data)
    predicted_plot[:, :] = np.nan
    predicted_plot[window_size:] = all_predictions

    plt.figure(figsize=(16, 8))
    plt.title("LSTM Model - Kaggle Weather Temperature Prediction")
    plt.xlabel("Date")
    plt.ylabel(temperature_column)
    plt.plot(df_weather.index, data, label="Actual Temperature", color="blue")
    plt.plot(
        df_weather.index,
        predicted_plot,
        label="Model Prediction",
        color="orange",
        alpha=0.8,
    )
    plt.legend(loc="lower right")
    plt.show()

    last_window_data = scaled_data[-window_size:]
    last_window_data_reshaped = last_window_data.reshape(1, window_size, 1)

    predicted_scaled_temperature = model.predict(last_window_data_reshaped)
    predicted_temperature = scaler.inverse_transform(predicted_scaled_temperature)

    last_actual_temperature = scaler.inverse_transform(scaled_data[-1].reshape(-1, 1))

    print(f"Actual Mean Temperature (Last Day in Data): {last_actual_temperature[0][0]:.2f} C")
    print(f"Predicted Mean Temperature for the Next Day: {predicted_temperature[0][0]:.2f} C")
else:
    print("Skipping prediction and visualization: Model or scaled_data not available.")
```
